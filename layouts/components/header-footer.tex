% DESCRIPTION:
% Configures the design of the running headers and footers for the second 
% page onwards using the fancyhdr package.

\fancypagestyle{runningpages}{
  \fancyhf{}
  \fancyhead[LE,LO]{\@runningauthor}
  \fancyhead[RE,RO]{\@articletitle}
  \fancyfoot[LE]{\thepage\ | \@journalname}
  \fancyfoot[RO]{\@journalname\ | \thepage}
  \renewcommand{\headrulewidth}{0pt}
  % Add a line above footer
  \renewcommand{\footrulewidth}{0.5pt}
}
\pagestyle{runningpages}
